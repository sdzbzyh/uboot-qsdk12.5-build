#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <net/tcp.h>

#define CHECK(x) do { \
    if (!(x)) { \
        printf("FAIL line=%d: %s\n", __LINE__, #x); \
        failures++; \
    } else { \
        passes++; \
    } \
} while (0)

static int passes, failures;
static unsigned emitted;
static u8 last_packet[1600];
static u16 last_len;
static u8 last_src[4], last_dst[4];

static const u8 server_ip[4] = {10,0,0,1};
static const u8 client_ip[4] = {10,0,0,2};

static int emit_cb(void *arg, const u8 src[4], const u8 dst[4],
                   const u8 *packet, u16 len)
{
    (void)arg;
    if (len > sizeof(last_packet))
        return -1;
    memcpy(last_packet, packet, len);
    memcpy(last_src, src, 4);
    memcpy(last_dst, dst, 4);
    last_len = len;
    emitted++;
    return 0;
}

static void put16(u8 *p, u16 v)
{
    p[0] = v >> 8;
    p[1] = v;
}

static void put32(u8 *p, u32 v)
{
    p[0] = v >> 24;
    p[1] = v >> 16;
    p[2] = v >> 8;
    p[3] = v;
}


static u32 get32(const u8 *p)
{
    return ((u32)p[0] << 24) |
           ((u32)p[1] << 16) |
           ((u32)p[2] << 8) | p[3];
}

static u16 make_packet(u8 *p, u16 sport, u16 dport,
                       u32 seq, u32 ack, u8 flags,
                       const u8 *data, u16 len)
{
    memset(p, 0, 20 + len);
    put16(p, sport);
    put16(p + 2, dport);
    put32(p + 4, seq);
    put32(p + 8, ack);
    p[12] = 0x50;
    p[13] = flags;
    put16(p + 14, 4096);
    if (len)
        memcpy(p + 20, data, len);
    put16(p + 16, tcp_checksum_ipv4(client_ip, server_ip,
                                    p, 20 + len));
    return 20 + len;
}

static int input(struct tcp_core_connection *c, u32 seq,
                 u32 ack, u8 flags, const u8 *data,
                 u16 len, u32 now)
{
    u8 packet[1600];
    u16 plen = make_packet(packet, 40000, 8080,
                           seq, ack, flags, data, len);
    return tcp_core_input(c, client_ip, server_ip,
                          packet, plen, now, emit_cb, NULL);
}

static void handshake(struct tcp_core_connection *c)
{
    CHECK(tcp_core_listen(c, server_ip, 8080, 1000) == 0);

    CHECK(input(c, 5000, 0, TCP_FLAG_SYN, NULL, 0, 1) == 1);
    CHECK(c->state == TCP_CORE_SYN_RECEIVED);
    CHECK((last_packet[13] & 0x12) == 0x12);
    CHECK(get32(last_packet + 4) == 1000);
    CHECK(get32(last_packet + 8) == 5001);

    CHECK(input(c, 5001, 1001, TCP_FLAG_ACK, NULL, 0, 2) == 1);
    CHECK(c->state == TCP_CORE_ESTABLISHED);
    CHECK(c->snd_una == 1001);
    CHECK(c->rcv_nxt == 5001);
}

static void test_data_and_ack(void)
{
    struct tcp_core_connection c;
    u8 received[16];

    printf("\n[DATA AND CUMULATIVE ACK]\n");
    handshake(&c);

    CHECK(input(&c, 5001, 1001,
                TCP_FLAG_ACK | TCP_FLAG_PSH,
                (const u8 *)"abc", 3, 3) == 1);
    CHECK(c.rx_len == 3);
    CHECK(c.rcv_nxt == 5004);
    CHECK((last_packet[13] & TCP_FLAG_ACK) != 0);
    CHECK(get32(last_packet + 8) == 5004);

    CHECK(tcp_core_read(&c, received, sizeof(received)) == 3);
    CHECK(memcmp(received, "abc", 3) == 0);
    CHECK(c.rx_len == 0);

    CHECK(tcp_core_write(&c, (const u8 *)"reply", 5,
                         4, emit_cb, NULL) == 5);
    CHECK(c.pending);
    CHECK(c.snd_nxt == 1006);
    CHECK(last_packet[13] & TCP_FLAG_PSH);

    CHECK(input(&c, 5004, 1006, TCP_FLAG_ACK,
                NULL, 0, 5) == 1);
    CHECK(!c.pending);
    CHECK(c.snd_una == 1006);
}

static void test_retransmission(void)
{
    struct tcp_core_connection c;
    unsigned initial;

    printf("\n[RETRANSMISSION]\n");
    handshake(&c);
    initial = emitted;

    CHECK(tcp_core_write(&c, (const u8 *)"retry", 5,
                         10, emit_cb, NULL) == 5);

    CHECK(tcp_core_poll(&c, 1009, emit_cb, NULL) == 0);
    CHECK(tcp_core_poll(&c, 1010, emit_cb, NULL) == 1);
    CHECK(c.retries == 1);

    CHECK(tcp_core_poll(&c, 2010, emit_cb, NULL) == 1);
    CHECK(tcp_core_poll(&c, 3010, emit_cb, NULL) == 1);
    CHECK(tcp_core_poll(&c, 4010, emit_cb, NULL) == 1);
    CHECK(tcp_core_poll(&c, 5010, emit_cb, NULL) == 1);
    CHECK(c.retries == TCP_CORE_MAX_RETRIES);
    CHECK(emitted == initial + 6);

    CHECK(tcp_core_poll(&c, 6010, emit_cb, NULL) == 1);
    CHECK(c.state == TCP_CORE_LISTEN);
}

static void test_close_wait_data(void)
{
    struct tcp_core_connection c;

    printf("\n[CLOSE-WAIT AFTER FIN]\n");
    handshake(&c);

    CHECK(input(&c, 5001, 1001,
                TCP_FLAG_FIN | TCP_FLAG_ACK,
                NULL, 0, 10) == 1);
    CHECK(c.state == TCP_CORE_CLOSE_WAIT);
    CHECK(c.rcv_nxt == 5002);

    CHECK(input(&c, 5002, 1001,
                TCP_FLAG_ACK | TCP_FLAG_PSH,
                (const u8 *)"X", 1, 11) == 1);

    printf("OBSERVE_CLOSE_WAIT_DATA: state=%d rx_len=%u rcv_nxt=%u\n",
           c.state, c.rx_len, c.rcv_nxt);

    if (c.rx_len == 1 && c.rx_data[0] == 'X')
        printf("CONFIRMED: data accepted after peer FIN\n");
    else
        printf("NOT_REPRODUCED: post-FIN data rejected\n");
}

static void test_time_wait_duplicate_fin(void)
{
    struct tcp_core_connection c;
    u32 original_deadline;
    unsigned before;

    printf("\n[TIME-WAIT DUPLICATE FIN]\n");
    handshake(&c);

    CHECK(tcp_core_close(&c, 10, emit_cb, NULL) == 0);
    CHECK(c.state == TCP_CORE_FIN_WAIT_1);

    CHECK(input(&c, 5001, 1002, TCP_FLAG_ACK,
                NULL, 0, 11) == 1);
    CHECK(c.state == TCP_CORE_FIN_WAIT_2);

    CHECK(input(&c, 5001, 1002,
                TCP_FLAG_FIN | TCP_FLAG_ACK,
                NULL, 0, 12) == 1);
    CHECK(c.state == TCP_CORE_TIME_WAIT);

    original_deadline = c.time_wait_deadline;
    before = emitted;

    CHECK(input(&c, 5001, 1002,
                TCP_FLAG_FIN | TCP_FLAG_ACK,
                NULL, 0, 20000) == 1);

    CHECK(emitted == before + 1);
    CHECK(last_packet[13] & TCP_FLAG_ACK);
    CHECK(get32(last_packet + 8) == 5002);

    printf("ORIGINAL_DEADLINE=%u\n", original_deadline);
    printf("AFTER_DUPLICATE_FIN=%u\n", c.time_wait_deadline);

    if (c.time_wait_deadline == original_deadline)
        printf("CONFIRMED: duplicate FIN did not restart TIME-WAIT\n");
    else
        printf("TIME-WAIT deadline changed\n");
}
