static void test_time_wait_expiry(void)
{
    struct tcp_core_connection c;

    printf("\n[TIME-WAIT EXPIRY]\n");
    handshake(&c);
    CHECK(tcp_core_close(&c, 10, emit_cb, NULL) == 0);
    CHECK(input(&c, 5001, 1002, TCP_FLAG_ACK,
                NULL, 0, 11) == 1);
    CHECK(input(&c, 5001, 1002,
                TCP_FLAG_FIN | TCP_FLAG_ACK,
                NULL, 0, 12) == 1);

    CHECK(c.state == TCP_CORE_TIME_WAIT);
    CHECK(tcp_core_poll(&c, c.time_wait_deadline - 1,
                        emit_cb, NULL) == 0);
    CHECK(c.state == TCP_CORE_TIME_WAIT);
    CHECK(tcp_core_poll(&c, c.time_wait_deadline,
                        emit_cb, NULL) == 0);
    CHECK(c.state == TCP_CORE_LISTEN);
}

static void test_tuple_sequence_and_ack(void)
{
    struct tcp_core_connection c;
    u8 packet[64];
    u16 n;

    printf("\n[TUPLE SEQUENCE ACK AND INVALID FLAGS]\n");
    handshake(&c);

    n = make_packet(packet, 40001, 8080, 5001, 1001,
                    TCP_FLAG_ACK, NULL, 0);
    CHECK(tcp_core_input(&c, client_ip, server_ip, packet, n,
                         10, emit_cb, NULL) == 0);
    CHECK(c.state == TCP_CORE_ESTABLISHED);

    CHECK(input(&c, 5002, 1001,
                TCP_FLAG_ACK | TCP_FLAG_PSH,
                (const u8 *)"Z", 1, 11) == 1);
    CHECK(c.rx_len == 0);
    CHECK(get32(last_packet + 8) == 5001);

    CHECK(input(&c, 5001, 999999,
                TCP_FLAG_ACK, NULL, 0, 12) == 0);
    CHECK(c.state == TCP_CORE_ESTABLISHED);

    CHECK(tcp_core_listen(&c, server_ip, 8080, 1000) == 0);
    CHECK(input(&c, 5000, 1001,
                TCP_FLAG_SYN | TCP_FLAG_ACK,
                NULL, 0, 13) == 0);
    CHECK(c.state == TCP_CORE_LISTEN);
}

static void test_buffer_boundary_and_malformed(void)
{
    struct tcp_core_connection c;
    u8 data[512], packet[600];
    unsigned i;
    u16 n;

    printf("\n[RX BUFFER BOUNDARY AND MALFORMED INPUT]\n");
    memset(data, 'B', sizeof(data));
    handshake(&c);

    for (i = 0; i < 8; i++) {
        CHECK(input(&c, 5001 + i * 512, 1001,
                    TCP_FLAG_ACK | TCP_FLAG_PSH,
                    data, sizeof(data), 20 + i) == 1);
    }
    CHECK(c.rx_len == TCP_CORE_RX_CAPACITY);
    CHECK(c.rcv_nxt == 5001 + TCP_CORE_RX_CAPACITY);

    CHECK(input(&c, c.rcv_nxt, 1001,
                TCP_FLAG_ACK | TCP_FLAG_PSH,
                (const u8 *)"Q", 1, 40) == 1);
    CHECK(c.rx_len == TCP_CORE_RX_CAPACITY);

    n = make_packet(packet, 40000, 8080, c.rcv_nxt, 1001,
                    TCP_FLAG_ACK, NULL, 0);
    CHECK(tcp_core_input(&c, client_ip, server_ip, packet, 10,
                         41, emit_cb, NULL) == -1);

    packet[12] = 0x40;
    n = make_packet(packet, 40000, 8080, c.rcv_nxt, 1001,
                    TCP_FLAG_ACK, NULL, 0);
    packet[12] = 0x40;
    CHECK(tcp_core_input(&c, client_ip, server_ip, packet, n,
                         42, emit_cb, NULL) == -1);

    n = make_packet(packet, 40000, 8080, c.rcv_nxt, 1001,
                    TCP_FLAG_ACK, NULL, 0);
    packet[16] ^= 0x80;
    CHECK(tcp_core_input(&c, client_ip, server_ip, packet, n,
                         43, emit_cb, NULL) == -1);
}
